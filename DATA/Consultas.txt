**************************************************************************************
******************CONSULTA UNIENDO COLECCIONES JOIN***********************************
**************************************************************************************

db.IPS.aggregate([
  // Unir con UBICACION
  {
    $lookup: {
      from: "ubicaciones",
      localField: "DANE",
      foreignField: "_id",
      as: "info_ubicacion"
    }
  },
  // Unir con TIPO
  {
    $lookup: {
      from: "tipos_IPS",
      localField: "TIPO",
      foreignField: "_id",
      as: "info_tipo"
    }
  },
  // Unir con CALIFICACION
  {
    $lookup: {
      from: "calificaciones",
      localField: "CALIFICACION",
      foreignField: "_id",
      as: "info_calificacion"
    }
  },
  // Desempaquetar los arrays resultantes (preservando nulos si algún código no cruza)
  { $unwind: { path: "$info_ubicacion", preserveNullAndEmptyArrays: true } },
  { $unwind: { path: "$info_tipo", preserveNullAndEmptyArrays: true } },
  { $unwind: { path: "$info_calificacion", preserveNullAndEmptyArrays: true } },
  // Dar formato limpio a la salida
  {
    $project: {
      _id: 0,
      NIT: 1,
      Razon_Social: 1,
      Municipio: "$info_ubicacion.municipio",
      Departamento: "$info_ubicacion.departamento",
      Tipo_IPS: "$info_tipo.TIPO",
      Calificacion: "$info_calificacion.CALIFICACION",
      Estado: { $cond: [{ $eq: ["$Activa", 1] }, "Activa", "Inactiva"] }
    }
  }
]);

//Contar dos valores distintos y sumarlos (ej. IPS Públicas + IPS Privadas)

db.IPS.aggregate([
  {
    $group: {
      _id: null,
      total_publicas: { $sum: { $cond: [{ $eq: ["$TIPO", 4] }, 1, 0] } },
      total_privadas: { $sum: { $cond: [{ $eq: ["$TIPO", 1] }, 1, 0] } }
    }
  },
  {
    $project: {
      _id: 0,
      total_publicas: 1,
      total_privadas: 1,
      suma_publicas_y_privadas: { $add: ["$total_publicas", "$total_privadas"] }
    }
  }
]);


**************************************************************************************
*******************************AGRUPAR REGISTROS**************************************
**************************************************************************************

db.IPS.aggregate([
  // 1. Filtrar Activa = 1 y CALIFICACION = 4
  {
    $match: {
      Activa: 1,
      CALIFICACION: 4
    }
  },

  // 2. Cruzar con la colección exacta "TIPO"
  {
    $lookup: {
      from: "TIPO",
      localField: "TIPO",
      foreignField: "ID",   // <-- Si en db.TIPO.findOne() se llama "_id", cámbialo por "_id"
      as: "tipo_doc"
    }
  },
  { $unwind: "$tipo_doc" },

  // 3. Agrupar y calcular métricas
  {
    $group: {
      _id: "$tipo_doc.TIPO",
      total_IPS: { $sum: 1 },
      total_activas: { $sum: "$Activa" },
      total_inactivas: { $sum: { $subtract: [1, "$Activa"] } },
      promedio_calificacion: { $avg: "$CALIFICACION" }
    }
  },

  // 4. Proyectar resultados finales
  {
    $project: {
      _id: 0,
      Tipo_IPS: "$_id",
      total_IPS: 1,
      total_activas: 1,
      total_inactivas: 1,
      promedio_calificacion: { $round: ["$promedio_calificacion", 4] },
      porcentaje_activas: {
        $round: [
          { $multiply: [{ $divide: ["$total_activas", "$total_IPS"] }, 100] },
          1
        ]
      }
    }
  }
]);
